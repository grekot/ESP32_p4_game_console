// ROZWIAZANIE zadan 4-5 z lekcji 03: przyspieszanie, zycia, GAME OVER, restart.
#include "console/console.h"

namespace {

const int PADDLE_W = 120;
const int PADDLE_H = 16;
const int PADDLE_Y = 440;
const int RADIUS   = 12;

int paddle_x = 340;
int ball_x   = 400;
int ball_y   = 0;
int speed    = 7;
int score    = 0;
int misses   = 0;
int lives    = 3;      // zad. 5

void setup()
{
    paddle_x = 340;
    ball_x   = random(RADIUS, screen_width() - RADIUS);
    ball_y   = 0;
    speed    = 7;
    score    = 0;
    misses   = 0;
    lives    = 3;
}

void frame()
{
    clear(DARK_GREEN);

    if (lives > 0) {
        // --- gra trwa ---
        if (held(LEFT)) {
            paddle_x = paddle_x - 8;
        }
        if (held(RIGHT)) {
            paddle_x = paddle_x + 8;
        }
        paddle_x = clamp(paddle_x, 0, screen_width() - PADDLE_W);

        ball_y = ball_y + speed;

        if (ball_y + RADIUS >= PADDLE_Y && ball_x >= paddle_x && ball_x <= paddle_x + PADDLE_W) {
            score = score + 1;
            speed = clamp(speed + 1, 5, 20);   // zad. 4: szybciej, ale nie bez konca
            ball_y = 0;
            ball_x = random(RADIUS, screen_width() - RADIUS);
        }

        if (ball_y > screen_height()) {
            misses = misses + 1;
            lives  = lives - 1;                // zad. 5
            ball_y = 0;
            ball_x = random(RADIUS, screen_width() - RADIUS);
        }
    } else {
        // --- koniec gry ---
        text(240, 190, "GAME OVER", RED, 4);
        text(300, 260, "Wcisnij A", WHITE, 2);
        if (pressed(A)) {
            restart();                         // od nastepnej klatki setup() jeszcze raz
        }
    }

    rect(paddle_x, PADDLE_Y, PADDLE_W, PADDLE_H, WHITE);
    circle(ball_x, ball_y, RADIUS, ORANGE);
    text(20, 16, score, WHITE, 3);
    text(720, 16, lives, RED, 3);

    watch("paddle_x", paddle_x);
    watch("score", score);
    watch("misses", misses);
    watch("lives", lives);
    watch("speed", speed);
}

}  // namespace

CONSOLE_ADD_GAME(lapacz, "Lapacz", "Lekcja 03: klawisze i warunki")
